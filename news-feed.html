function renderFeed(stories) { const container = document.getElementById('news-container'); container.innerHTML = ''; if (!stories || stories.length === 0) { container.innerHTML = `

No recent articles found matching the byline filter.

`; return; } stories.forEach(story => { const card = document.createElement('article'); card.className = 'news-card'; card.innerHTML = `
${story.source || 'News'} ${story.pubDate ? new Date(story.pubDate).toLocaleDateString() : ''}

${story.title}

${story.snippet || ''}

Read full story → `; container.appendChild(card); }); }
